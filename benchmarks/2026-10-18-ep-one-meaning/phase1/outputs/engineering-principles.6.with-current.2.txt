## scripts/api.py (new file)

```python
import logging
import time

import requests

BASE_URL = "https://api.newhost.io/v2"  # single source of truth for the API host

log = logging.getLogger(__name__)


def get(path, retries=3, backoff=1.0, timeout=10):
    """GET BASE_URL + path, retrying on network errors and non-2xx responses."""
    url = BASE_URL + path
    for attempt in range(1, retries + 1):
        try:
            log.info("GET %s (attempt %d/%d)", url, attempt, retries)
            resp = requests.get(url, timeout=timeout)
            resp.raise_for_status()
            log.info("GET %s -> %d", url, resp.status_code)
            return resp
        except requests.RequestException as exc:
            log.warning("GET %s failed: %s", url, exc)
            if attempt == retries:
                log.error("GET %s giving up after %d attempts", url, retries)
                raise
            time.sleep(backoff * 2 ** (attempt - 1))
```

## Notes

- `from api import get` works when you run `python scripts/fetch_users.py`, because the script's folder is on the import path. If you run the scripts as modules (`python -m scripts.fetch_users`), change it to `from scripts.api import get`.

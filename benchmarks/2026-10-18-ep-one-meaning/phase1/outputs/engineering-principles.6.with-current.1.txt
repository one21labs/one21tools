## scripts/api.py (new file)

```python
import logging
import time

import requests

BASE = "https://api.newhost.io/v2"  # single source of truth for the API URL

log = logging.getLogger("api")

RETRIES = 3
BACKOFF_SECONDS = 1


def get(path):
    url = BASE + path
    for attempt in range(1, RETRIES + 1):
        try:
            log.info("GET %s (attempt %d/%d)", url, attempt, RETRIES)
            resp = requests.get(url, timeout=10)
            resp.raise_for_status()
            log.info("GET %s -> %d", url, resp.status_code)
            return resp
        except requests.RequestException as exc:
            log.warning("GET %s failed: %s", url, exc)
            if attempt == RETRIES:
                log.error("giving up on %s after %d attempts", url, RETRIES)
                raise
            time.sleep(BACKOFF_SECONDS * attempt)
```
